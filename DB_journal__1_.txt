# ================================================= # 3. BACKUP ONLY DATABASE STRUCTURE
# Only CREATE TABLE statements
# =================================================
def structure_backup():
conn = connect_database() cursor = conn.cursor()

# Get CREATE TABLE commands
cursor.execute(""" SELECT sql
FROM sqlite_master WHERE type='table'
AND name NOT LIKE 'sqlite_%' """)
tables = cursor.fetchall() # Write structure into file
with open("structure_backup.sql", "w") as file:
for table in tables: file.write(table[0] + ";\n")
conn.close()
print("Database structure backup created successfully.")

# ================================================= # 4. BACKUP ONLY DATA
# Only INSERT statements
# =================================================
def data_backup():

conn = connect_database() cursor = conn.cursor()
# Find all table names cursor.execute(""" SELECT name FROM sqlite_master WHERE type='table'
AND name NOT LIKE 'sqlite_%' """)
tables = cursor.fetchall()
with open("data_backup.sql", "w") as file: # Loop through every table
for table in tables:
table_name = table[0]

# Read all records from table cursor.execute(f"SELECT * FROM {table_name}")
rows = cursor.fetchall()
# Convert each record into INSERT query for row in rows:

values = []
for value in row:
# If value is text
if isinstance(value, str):
value = value.replace("'", "''")
values.append(f"'{value}'")
# If value is NULL elif value is None:
values.append("NULL")
# If value is number else:
values.append(str(value)) values_text = ", ".join(values)
file.write(
f"INSERT INTO {table_name} VALUES ({values_text});\n"
)
conn.close()
print("Data backup created successfully.")

# ================================================= # 5. RESTORE DATABASE FROM BACKUP
# =================================================
def restore_database():

# Create/open another database conn = sqlite3.connect("restored.db")
# Open complete backup file
with open("complete_backup.sql", "r") as file: sql_script = file.read()
try:

# Run all SQL commands from backup file conn.executescript(sql_script)
conn.commit()

print("Database restored successfully.")

except sqlite3.Error as error: print("Restore Error:", error)
finally:
conn.close()

# ================================================= # 6. EXPORT COMPLETE DATABASE TO ANOTHER FILE

]

for line in conn.iterdump(): file.write(line + "\n")
print("Complete backup created successfully.") conn.close()
# Step 5: Create a NEW database

new_conn = sqlite3.connect("company_restore.db") # Step 6: Read the backup SQL file
with open("company_backup.sql", "r") as file: backup_sql = file.read()
